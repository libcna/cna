cmake_minimum_required(VERSION 3.20)
project(CnaSharedRuntimeProbe LANGUAGES CXX)
include("${CNA_SOURCE_DIR}/cmake/SharedRuntimePolicy.cmake")
include("${CNA_SOURCE_DIR}/cmake/SharedRuntimeLibrary.cmake")
if(PROBE_DIRECT_HELPER)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
if(POLICY CMP0156)
    # General linker-capability deduplication is not what fixes whole-archive cycles.
    cmake_policy(SET CMP0156 OLD)
endif()
set(CNA_SHARED_MODULES_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(probe_a STATIC a.cpp)
add_library(probe_b STATIC b.cpp)
target_link_libraries(probe_a PUBLIC probe_b)
target_link_libraries(probe_b PUBLIC probe_a)
add_library(CNA INTERFACE)
if(CNA_SHARED_LIBRARY OR PROBE_DIRECT_HELPER)
    cna_define_shared_runtime(CNA probe_a)
else()
    target_link_libraries(CNA INTERFACE probe_a)
endif()
add_executable(shared_runtime_probe main.cpp)
target_link_libraries(shared_runtime_probe PRIVATE CNA)
foreach(_target IN ITEMS probe_a probe_b cna_shared shared_runtime_probe)
    if(TARGET ${_target})
        target_compile_options(${_target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/selected-mode.txt" "${CNA_SHARED_LIBRARY}\n")
