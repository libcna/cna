# cna-engine UP-001: a source-independent consumer of the installed runtime.
cmake_minimum_required(VERSION 3.20)
if(PROBE_C_ONLY)
    project(CnaInstalledRuntimeProbe LANGUAGES C)
else()
    project(CnaInstalledRuntimeProbe LANGUAGES C CXX)
endif()
# A cross toolchain can overwrite CMAKE_FIND_ROOT_PATH during project(). Preserve the moved SDK
# in the consumer search roots AFTER the toolchain has run, without changing the producer.
if(PROBE_FIND_ROOT)
    list(PREPEND CMAKE_FIND_ROOT_PATH "${PROBE_FIND_ROOT}")
endif()
enable_testing()
if(PROBE_CASE)
    include(PackageCases.cmake)
    return()
endif()

function(find_sdk)
    # Custom link features must survive a consumer's function scope too.
    find_package(CNA 0.1 CONFIG REQUIRED)
endfunction()
if(PROBE_STATIC_MIXED AND PROBE_C_ONLY)
    set(CMAKE_DISABLE_FIND_PACKAGE_SharpRuntime TRUE)
    find_package(CNA 0.1 CONFIG REQUIRED COMPONENTS CApi)
else()
    find_sdk()
    find_package(CNA 0.1 CONFIG REQUIRED)
endif()
# GAP-13: archive groups must retain installed target identities. A bare build target inside
# LINK_GROUP becomes an unresolved -l name after export, even when its archive was shipped.
if(MINGW AND TARGET CNA::DependencySdlShadercross)
    get_target_property(_cross_links CNA::DependencySdlShadercross INTERFACE_LINK_LIBRARIES)
    string(REGEX MATCH "LINK_GROUP:RESCAN,([^>]+)" _cross_group "${_cross_links}")
    if(NOT _cross_group)
        message(FATAL_ERROR "Installed ShaderCross lost its archive rescan group")
    endif()
    string(REPLACE "," ";" _cross_members "${CMAKE_MATCH_1}")
    foreach(_member IN LISTS _cross_members)
        if(NOT TARGET "${_member}")
            message(FATAL_ERROR "Installed ShaderCross group refers to a non-exported target: ${_member}")
        endif()
    endforeach()
endif()
if(PROBE_EXPECT_STATIC AND NOT TARGET CNA::CApiStatic)
    message(FATAL_ERROR "Configured static C archive is missing from the moved SDK")
endif()
option(PROBE_STATIC_MIXED "Require refusal of the encapsulated static C/C++ composition" OFF)
if(PROBE_STATIC_MIXED)
    if(NOT TARGET CNA::CApiStatic OR NOT TARGET "${PROBE_MIXED_TARGET}")
        message(FATAL_ERROR "Static mixed probe requires both installed targets")
    endif()
    add_executable(InstalledStaticMixed api.c)
    if(PROBE_MIXED_ORDER STREQUAL "CFirst")
        target_link_libraries(InstalledStaticMixed PRIVATE CNA::CApiStatic ${PROBE_MIXED_TARGET})
    else()
        target_link_libraries(InstalledStaticMixed PRIVATE ${PROBE_MIXED_TARGET} CNA::CApiStatic)
    endif()
    return()
endif()
if(NOT COMMAND cna_deploy_runtime)
    message(FATAL_ERROR "Installed CNA package does not export cna_deploy_runtime()")
endif()
set(_runtime_targets
    CNA::Core CNA::Math CNA::Input CNA::GraphicsCore
    CNA::Content CNA::Runtime CNA::Audio
    SharpRuntime::Core.Base SharpRuntime::IO)
set(_missing)
foreach(_target IN LISTS _runtime_targets)
    if(NOT TARGET ${_target})
        list(APPEND _missing ${_target})
    endif()
endforeach()
if(_missing)
    message(FATAL_ERROR "Missing exported runtime targets: ${_missing}")
endif()

foreach(_target IN ITEMS InstalledMath InstalledMathStaticCpp)
    add_executable(${_target} math.cpp)
    target_link_libraries(${_target} PRIVATE CNA::Math)
    if(NOT MSVC)
        target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${_target} COMMAND ${_target})
endforeach()
if(CMAKE_EXECUTABLE_FORMAT STREQUAL "ELF" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_link_options(InstalledMathStaticCpp PRIVATE -static-libstdc++ -static-libgcc)
    find_program(_readelf NAMES readelf llvm-readelf REQUIRED)
    foreach(_target IN ITEMS InstalledMath InstalledMathStaticCpp)
        add_test(NAME ${_target}_Closure COMMAND ${CMAKE_COMMAND}
            -DPROBE=$<TARGET_FILE:${_target}> -DREADELF=${_readelf}
            -DSTATIC_CPP=$<STREQUAL:${_target},InstalledMathStaticCpp>
            -P "${CMAKE_CURRENT_SOURCE_DIR}/CheckMathClosure.cmake")
    endforeach()
endif()
foreach(_order IN ITEMS CppFirst CFirst)
    add_executable(InstalledRuntime_${_order} main.cpp)
    if(TARGET CNA::CApi AND NOT PROBE_CPP_ONLY AND _order STREQUAL "CFirst")
        target_link_libraries(InstalledRuntime_${_order} PRIVATE CNA::CApi)
    endif()
    target_link_libraries(InstalledRuntime_${_order} PRIVATE ${_runtime_targets})
    cna_deploy_runtime(InstalledRuntime_${_order})
    if(TARGET CNA::CApi AND NOT PROBE_CPP_ONLY)
        target_sources(InstalledRuntime_${_order} PRIVATE mixed.c)
        target_compile_definitions(InstalledRuntime_${_order} PRIVATE CNA_PACKAGE_TEST_C_API)
        if(_order STREQUAL "CppFirst")
            target_link_libraries(InstalledRuntime_${_order} PRIVATE CNA::CApi)
        endif()
    endif()
    if(MSVC)
        target_compile_options(InstalledRuntime_${_order} PRIVATE /W4 /WX)
    else()
        target_compile_options(InstalledRuntime_${_order} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    # The package publishes its default renderer, and the consumer's own translation units must
    # report the same one; for OPENGL33/WEBGL2 that needs the exported CNA_GL_PROFILE_* define.
    if(NOT CNA_SDK_GRAPHICS_RENDERER)
        message(FATAL_ERROR "The CNA package does not publish CNA_SDK_GRAPHICS_RENDERER")
    endif()
    target_compile_definitions(InstalledRuntime_${_order} PRIVATE
        CNA_PACKAGE_EXPECTED_RENDERER="${CNA_SDK_GRAPHICS_RENDERER}")
    add_test(NAME InstalledRuntime_${_order} COMMAND InstalledRuntime_${_order})
endforeach()
if(EMSCRIPTEN)
    # CNA-FIX-L19: name each installed export directly. These two Node consumers
    # suspend/resume a C++ stack; missing either export must fail generation.
    foreach(_contract IN ITEMS Asyncify Abi)
        add_executable(l19-${_contract} l19-emscripten.cpp)
        target_link_libraries(l19-${_contract} PRIVATE CNA::Math CNA::Emscripten${_contract})
        if(_contract STREQUAL "Asyncify")
            target_link_libraries(l19-${_contract} PRIVATE CNA::EmscriptenExceptionAbi)
        endif()
        add_test(NAME l19-${_contract} COMMAND l19-${_contract})
        set_tests_properties(l19-${_contract} PROPERTIES TIMEOUT 30)
    endforeach()
    # Run by the driver in headless Chrome through the private runner, not by Node/CTest.
    # Blocking Game::Run() needs the exported exception + Asyncify composition.
    add_executable(InstalledBrowserFrame browser.cpp)
    set_target_properties(InstalledBrowserFrame PROPERTIES SUFFIX ".html")
    target_compile_definitions(InstalledBrowserFrame PRIVATE
        PROBE_EXPECTED_RENDERER="${PROBE_EXPECTED_RENDERER}")
    target_compile_options(InstalledBrowserFrame PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(InstalledBrowserFrame PRIVATE
        CNA::CNA CNA::EmscriptenAsyncify CNA::EmscriptenAbi)
    target_link_options(InstalledBrowserFrame PRIVATE -sALLOW_MEMORY_GROWTH=1)
    cna_deploy_runtime(InstalledBrowserFrame)
endif()

# GAP-13: opt-in graphics qualification against the moved SDK, with producer-derived expectations.
if(PROBE_COMPILED_EFFECTS)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CnaConformanceEffect.fxb" _effect_hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," _effect_bytes "${_effect_hex}")
    # These generated literals belong to the copied test fixture. Keeping its headers together
    # preserves the audit requirement that explicit include paths come from the moved SDK.
    file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/compiled_effect_bytes.hpp"
        "// SPDX-License-Identifier: MS-PL\n#pragma once\ninline constexpr unsigned char CompiledEffectBytes[] = {${_effect_bytes}};\n")
    if(EMSCRIPTEN)
        message(FATAL_ERROR "GAP-13 browser package qualification is reserved for Phase 3")
    endif()
    add_executable(InstalledCompiledEffect compiled_effect.cpp)
    target_link_libraries(InstalledCompiledEffect PRIVATE CNA::CNA)
    cna_deploy_runtime(InstalledCompiledEffect)
    if(MINGW AND PROBE_COMPILED_LAUNCHER)
        add_test(NAME InstalledCompiledEffect COMMAND "${PROBE_COMPILED_LAUNCHER}"
            "$<TARGET_FILE:InstalledCompiledEffect>")
    else()
        add_test(NAME InstalledCompiledEffect COMMAND InstalledCompiledEffect)
    endif()
    set_tests_properties(InstalledCompiledEffect PROPERTIES
        ENVIRONMENT "CNA_GRAPHICS_RENDERER=${PROBE_COMPILED_RENDERER};SDL_AUDIODRIVER=dummy")
    set(_compiled_target InstalledCompiledEffect)
    target_compile_definitions(${_compiled_target} PRIVATE
        PROBE_COMPILED_EFFECTS
        PROBE_EXPECT_COMPILED_EFFECTS=$<BOOL:${PROBE_EXPECT_COMPILED_EFFECTS}>)
    if(NOT EMSCRIPTEN)
        target_compile_definitions(${_compiled_target} PRIVATE
            PROBE_EXPECTED_RENDERER="${PROBE_COMPILED_RENDERER}")
    endif()
endif()
