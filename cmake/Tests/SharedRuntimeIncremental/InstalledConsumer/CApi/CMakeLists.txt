cmake_minimum_required(VERSION 3.20)
project(SharedRuntimeInstalledCApiConsumer LANGUAGES C)
find_package(CNA CONFIG REQUIRED COMPONENTS CApi CApiStatic)
if(TARGET CNA::Shared)
    message(FATAL_ERROR "The C-only package unexpectedly imports the C++ runtime target")
endif()
foreach(_kind IN ITEMS shared static)
    add_executable(installed_c_${_kind} main.c)
    set_target_properties(installed_c_${_kind} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
endforeach()
target_link_libraries(installed_c_shared PRIVATE CNA::CApi)
target_link_libraries(installed_c_static PRIVATE CNA::CApiStatic)
get_target_property(_static_links CNA::CApiStatic INTERFACE_LINK_LIBRARIES)
foreach(_link IN LISTS _static_links)
    if(_link MATCHES "^(cna|-lcna|CNA::Shared)$" OR _link MATCHES "libcna\\.so")
        message(FATAL_ERROR "Standalone static C package depends on the shared runtime: ${_link}")
    endif()
endforeach()
